cmake_minimum_required(VERSION 3.20)
project(PrisonDisplayWindows VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenSSL 3.0 REQUIRED)
add_library(prison_core src/protocol.cpp src/crypto.cpp)
target_include_directories(prison_core PUBLIC include)
target_link_libraries(prison_core PUBLIC OpenSSL::Crypto)
if(MSVC)
  target_compile_options(prison_core PUBLIC /utf-8 PRIVATE /W4 /permissive-)
else()
  target_compile_options(prison_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(WIN32)
  add_executable(prison_display WIN32 src/app_win32.cpp src/discovery_win32.cpp)
  target_link_libraries(prison_display PRIVATE prison_core user32 gdi32 ws2_32)
  target_include_directories(prison_display PRIVATE include)
  if(MSVC)
    target_compile_options(prison_display PRIVATE /utf-8 /W4 /permissive-)
  else()
    target_compile_options(prison_display PRIVATE -Wall -Wextra -Wpedantic)
    target_link_options(prison_display PRIVATE -municode)
  endif()
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(prison_core_tests tests/core_tests.cpp)
  target_link_libraries(prison_core_tests PRIVATE prison_core OpenSSL::SSL)
  add_test(NAME core COMMAND prison_core_tests "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
endif()
# No UI or installer is advertised until native capture/input and real Windows
# validation exist. This target deliberately builds only the portable core.
