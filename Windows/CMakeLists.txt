cmake_minimum_required(VERSION 3.20)
project(PrisonDisplayWindows VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenSSL 3.0 REQUIRED)
add_library(prison_core src/protocol.cpp src/crypto.cpp src/agent.cpp src/server.cpp src/pairing.cpp)
target_include_directories(prison_core PUBLIC include)
target_link_libraries(prison_core PUBLIC OpenSSL::SSL OpenSSL::Crypto)
if(WIN32)
  target_link_libraries(prison_core PUBLIC ws2_32 crypt32)
endif()
if(MSVC)
  target_compile_options(prison_core PUBLIC /utf-8 PRIVATE /W4 /permissive-)
else()
  target_compile_options(prison_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(WIN32)
  add_executable(prison_display WIN32 src/app_win32.cpp)
  target_link_libraries(prison_display PRIVATE prison_core user32 gdi32 ws2_32 crypt32 shell32 ole32)
  target_include_directories(prison_display PRIVATE include)
  if(MSVC)
    target_compile_options(prison_display PRIVATE /utf-8 /W4 /permissive-)
  else()
    target_compile_options(prison_display PRIVATE -Wall -Wextra -Wpedantic)
    target_link_options(prison_display PRIVATE -municode)
  endif()
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(prison_core_tests tests/core_tests.cpp)
  target_link_libraries(prison_core_tests PRIVATE prison_core OpenSSL::SSL)
  add_test(NAME core COMMAND prison_core_tests "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
  add_executable(prison_agent_tests tests/agent_tests.cpp)
  target_link_libraries(prison_agent_tests PRIVATE prison_core)
  add_test(NAME agent COMMAND prison_agent_tests)
endif()
# Simulated PC for end-to-end tests on macOS: the real agent and encrypted
# listener with a recording sink instead of SendInput.
if(NOT WIN32)
  add_executable(prison_agent_host tests/agent_host.cpp)
  target_link_libraries(prison_agent_host PRIVATE prison_core)
endif()
# The Windows app receives mouse and keyboard from a Mac. Screen capture and
# video are not implemented yet.
